'================================
' PROGRAM: ALG_QUAD
' PURPOSE: Quadratic roots
'================================

"ALG QUAD"→TITLE
Prog "CORE_FORMAT"

Input "A?",QA
Input "B?",QB
Input "C?",QC

If QA=0 Then
 0→VALID
 "A CANNOT BE 0"→ERRMSG
 Prog "CORE_ERROR"
 Return
IfEnd

QB^2-4*QA*QC→D
If D<0 Then
 0→VALID
 "NO REAL ROOTS"→ERRMSG
 Prog "CORE_ERROR"
 Return
IfEnd

(-QB+√D)/(2*QA)→R1
(-QB-√D)/(2*QA)→R2

Locate 1,6,"R1="
Locate 4,6,R1
Locate 1,7,"R2="
Locate 4,7,R2

Pause
Return
